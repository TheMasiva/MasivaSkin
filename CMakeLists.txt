cmake_minimum_required(VERSION 3.16)
project(MasivaSkin VERSION 0.9.12 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(MasivaProfile src/main.cpp)
if(WIN32)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "Masiva Skin requires x64")
    endif()
    add_library(MasivaSkinBridge SHARED src/windows/game_bridge.cpp)
    set_target_properties(MasivaSkinBridge PROPERTIES PREFIX "")
    target_compile_definitions(MasivaSkinBridge PRIVATE UNICODE _UNICODE NOMINMAX)
    target_link_libraries(MasivaSkinBridge PRIVATE shell32)
    enable_language(RC)
    add_executable(MasivaSkin WIN32 src/windows/presence.cpp src/windows/app.cpp src/windows/resources.rc)
    add_dependencies(MasivaSkin MasivaSkinBridge)
    target_include_directories(MasivaSkin PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/assets")
    target_compile_definitions(MasivaSkin PRIVATE UNICODE _UNICODE NOMINMAX)
    target_link_libraries(MasivaSkin PRIVATE user32 gdi32 comdlg32 shell32 gdiplus winhttp ws2_32 dwmapi)
    if(MINGW)
        target_link_options(MasivaSkinBridge PRIVATE -static -static-libgcc -static-libstdc++)
        target_link_options(MasivaSkin PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
    if(MSVC)
        target_link_options(MasivaSkin PRIVATE /MANIFEST:NO)
        target_compile_options(MasivaSkinBridge PRIVATE /W4 /utf-8 /permissive-)
        target_compile_options(MasivaSkin PRIVATE /W4 /utf-8 /permissive-)
    else()
        target_compile_options(MasivaSkinBridge PRIVATE -Wall -Wextra -Wpedantic)
        target_compile_options(MasivaSkin PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
